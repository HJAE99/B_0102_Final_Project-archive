docker pull postgres:16

docker volume create robotdata4_postgresql

docker run -d --name robotdb4_postgresql \
-e POSTGRES_DB=robotdb4_postgresql \
-e POSTGRES_USER=rokey \
-e POSTGRES_PASSWORD=rokey \
-p 5432:5432 \
-v robotdata4_postgresql:/var/lib/postgresql/data \
postgres:16

<- 사전 sql을 통한 스키마 생성 필요(DB컨테이너 내부에서 DB에 생성하는 경우)
sudo apt install postgresql-client
cd (호스트 pc) .slq 파일경로
docker exec -i robotdb4_postgresql psql -U rokey -d robotdb4_postgresql < .sql

docker pull redis:8.8

docker volume create robotdata4_redis

docker run -d --name robotdb4_redis \
-p 6379:6379 \
-v robotdata4_redis:/data \
redis:8.8 redis-server --appendonly yes \
--user rokey on ">rokey" ~* +@all
